Show me a safe retry helper in Kotlin.
Here's a battle-tested version using exponential backoff with jitter, so parallel clients don't stampede the API:
Kotlin
suspend fun retryWithBackoff(
maxAttempts: Int = 3,
baseDelay: Long = 2_000
): T {
repeat(maxAttempts) { attempt ->
try { return call() }
catch (e: IOException) {
if (attempt == maxAttempts - 1) throw e
}
delay(baseDelay * (attempt + 1)
+ Random.nextLong(250))
}
}
